\begin{tikzpicture}[

% ==========================
% 1. 入力データ (左側)
% ==========================

% Studentのレイヤー表現
\node[layer, fill=studentColor, minimum width=3cm] (s_l1) at (student_center) {};
\node[layer, fill=studentColor, minimum width=3cm, above=0.15cm of s_l1] (s_l2) {};

% 3. Teacher Model (右側下)
% ==========================

% Teacherのレイヤー表現
\node[layer, fill=teacherColor, minimum width=3cm] (t_l1) at (teacher_center) {};
\node[layer, fill=teacherColor, minimum width=3cm, above=0.15cm of t_l1] (t_l2) {};

% ==========================
% Student Frame Output
\node[above=1.5cm of student_box] (s_frame_base) {};

}

\node[above=1.5cm of teacher_box] (t_frame_base) {};
\foreach \x in {-1.2, -0.9, ..., 1.2} {

}

\draw[fill=studentColor, draw=none] ($(s_clip_base)+(-0.5,0)$) rectangle ++(0.3, 0.2);
